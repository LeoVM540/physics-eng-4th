\section{Angles between planes in cubic crystals\extra}
\[
\cos\varphi = \frac{h_1h_2+k_1k_2+l_1l_2}{\sqrt{(h_1^2+k_1^2+l_1^2)(h_2^2+k_2^2+l_2^2)}}:
\]
(100)/(010) $90^\circ$; (100)/(110) $45^\circ$; (100)/(111) $54.7^\circ$; (111)/($\bar1$11) $70.5^\circ$; (110)/(111) $35.3^\circ$.
